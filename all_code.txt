from src.interfaces.base_interfaces import IUnit, INeutralObject
from src.utils.logger import logger


class BaseUnit(IUnit):
    def __init__(self, name, unit_type, health, armor, attack, symbol, x=0, y=0, base=None):
        self.name = name
        self.unit_type = unit_type
        self.health = health
        self.armor = armor
        self.attack = attack
        self.symbol = symbol
        self.x = x
        self.y = y
        self.base = base

    def move(self, dx, dy, field):
        new_x = self.x + dx
        new_y = self.y + dy
        return field.move_unit(self, new_x, new_y)

    def take_damage(self, damage):
        real_damage = damage - (self.armor // 4)
        if real_damage < 1:
            real_damage = 1
        self.health = self.health - real_damage
        logger.log(f"юнит {self.name} получил урон: {real_damage}")

        if self.health <= 0:
            self.health = 0
            if self.base is not None:
                self.base.on_unit_destroyed(self)

    def get_info(self):
        return f"[{self.symbol}] {self.name} (тип: {self.unit_type}) | здоровье: {self.health}, броня: {self.armor}, атака: {self.attack} | координаты: ({self.x}, {self.y})"

    def __add__(self, other):
        # перегрузка оператора + для взаимодействия с нейтральным объектом
        if isinstance(other, INeutralObject):
            result = other.interact(self)
            logger.log(f"взаимодействие: {result}")
            return result
        return "нельзя сложить эти объекты"

from src.units.infantry import SwordRusher, BedBreaker
from src.units.archers import BowSpammer, FireballThrower
from src.units.cavalry import IronGolem, EnderRider
from src.utils.logger import logger


class BaseBuilding:
    def __init__(self, name="база красных (кровать)", x=1, y=1, health=500, max_units=6):
        self.name = name
        self.x = x
        self.y = y
        self.health = health
        self.max_units = max_units
        self.symbol = "B"
        self.units = []

    def create_unit(self, choice, field):
        if len(self.units) >= self.max_units:
            return False, f"ошибка: на базе уже максимум бойцов ({self.max_units})"

        spawn_pos = self.find_free_place(field)
        if spawn_pos is None:
            return False, "ошибка: вокруг кровати нет свободных клеток"

        sx = spawn_pos[0]
        sy = spawn_pos[1]

        if choice == "1":
            new_unit = SwordRusher(sx, sy, self)
        elif choice == "2":
            new_unit = BedBreaker(sx, sy, self)
        elif choice == "3":
            new_unit = BowSpammer(sx, sy, self)
        elif choice == "4":
            new_unit = FireballThrower(sx, sy, self)
        elif choice == "5":
            new_unit = IronGolem(sx, sy, self)
        elif choice == "6":
            new_unit = EnderRider(sx, sy, self)
        else:
            return False, "ошибка: неверный номер бойца"

        added = field.add_unit(new_unit, sx, sy)
        if added:
            self.units.append(new_unit)
            msg = f"создан юнит: {new_unit.name} на клетке ({sx}, {sy})"
            logger.log(msg)
            return True, msg
        else:
            return False, "ошибка: не удалось поставить юнита на поле"

    def find_free_place(self, field):
        steps = [(0, 1), (1, 0), (1, 1), (0, -1), (-1, 0), (2, 0), (0, 2)]
        for dx, dy in steps:
            nx = self.x + dx
            ny = self.y + dy
            if field.is_inside(nx, ny):
                if field.get_unit_at(nx, ny) is None:
                    if not (nx == self.x and ny == self.y):
                        return (nx, ny)
        return None

    def on_unit_destroyed(self, unit):
        if unit in self.units:
            self.units.remove(unit)
            logger.log(f"база зафиксировала гибель юнита: {unit.name}")

    def get_status(self):
        text = f"--- состояние базы: {self.name} ---\n"
        text += f"координаты кровати: ({self.x}, {self.y})\n"
        text += f"прочность кровати: {self.health}\n"
        text += f"количество юнитов: {len(self.units)} из {self.max_units}\n"

        if len(self.units) == 0:
            text += "список юнитов: пусто"
        else:
            text += "список юнитов:\n"
            for i in range(len(self.units)):
                text += f"  {i + 1}. {self.units[i].get_info()}\n"
        return text

import copy
import random
from src.world.landscapes import WoolBridgeLandscape, ObsidianPlatformLandscape, VoidGapLandscape
from src.world.neutral_objects import GoldenAppleCrate, DiamondGenerator, EmeraldGenerator, TntTrap
from src.utils.logger import logger


class GameField:
    def __init__(self, width=10, height=10, max_objects=18):
        self.width = width
        self.height = height
        self.max_objects = max_objects

        # создаем двумерный список для ландшафта
        self.grid_landscape = []
        for y in range(self.height):
            row = []
            for x in range(self.width):
                row.append(WoolBridgeLandscape())
            self.grid_landscape.append(row)

        self.base = None
        self.units = []
        self.neutral_objects = []

    def generate_default_world(self):
        random.seed(42)
        for y in range(self.height):
            for x in range(self.width):
                r = random.random()
                if r < 0.18:
                    self.grid_landscape[y][x] = ObsidianPlatformLandscape()
                elif r < 0.28:
                    self.grid_landscape[y][x] = VoidGapLandscape()
                else:
                    self.grid_landscape[y][x] = WoolBridgeLandscape()

        # добавляем 4 нейтральных объекта
        items = [
            GoldenAppleCrate(3, 2),
            DiamondGenerator(5, 4),
            EmeraldGenerator(2, 6),
            TntTrap(6, 3)
        ]
        for item in items:
            if self.is_inside(item.x, item.y):
                self.grid_landscape[item.y][item.x] = WoolBridgeLandscape()
                self.add_neutral_object(item)

    def total_objects_count(self):
        count = len(self.units) + len(self.neutral_objects)
        if self.base is not None:
            count = count + 1
        return count

    def is_inside(self, x, y):
        return 0 <= x < self.width and 0 <= y < self.height

    def set_base(self, base):
        if self.total_objects_count() >= self.max_objects:
            return False
        if not self.is_inside(base.x, base.y):
            return False
        self.grid_landscape[base.y][base.x] = ObsidianPlatformLandscape()
        # делаем клетки вокруг базы проходимыми мостами для спавна
        offsets = [(0, 1), (1, 0), (1, 1), (0, -1), (-1, 0), (2, 0), (0, 2)]
        for dx, dy in offsets:
            nx = base.x + dx
            ny = base.y + dy
            if self.is_inside(nx, ny):
                self.grid_landscape[ny][nx] = WoolBridgeLandscape()
        self.base = base
        logger.log(f"база поставлена в точке: ({base.x}, {base.y})")
        return True

    def add_unit(self, unit, x, y):
        if self.total_objects_count() >= self.max_objects:
            return False
        if not self.is_inside(x, y):
            return False
        if self.get_unit_at(x, y) is not None:
            return False
        if not self.grid_landscape[y][x].can_pass(unit):
            return False

        unit.x = x
        unit.y = y
        self.units.append(unit)
        return True

    def remove_unit(self, unit):
        if unit in self.units:
            self.units.remove(unit)
            if unit.base is not None:
                unit.base.on_unit_destroyed(unit)
            logger.log(f"юнит удален с поля: {unit.name}")
            return True
        return False

    def add_neutral_object(self, obj):
        if self.total_objects_count() >= self.max_objects:
            return False
        if not self.is_inside(obj.x, obj.y):
            return False
        self.neutral_objects.append(obj)
        return True

    def remove_neutral_object(self, obj):
        if obj in self.neutral_objects:
            self.neutral_objects.remove(obj)
            return True
        return False

    def get_unit_at(self, x, y):
        for unit in self.units:
            if unit.x == x and unit.y == y:
                return unit
        return None

    def get_neutral_at(self, x, y):
        for obj in self.neutral_objects:
            if obj.x == x and obj.y == y:
                return obj
        return None

    def move_unit(self, unit, new_x, new_y):
        if not self.is_inside(new_x, new_y):
            return False, "ошибка: нельзя выйти за границы карты"

        if self.base is not None and self.base.x == new_x and self.base.y == new_y:
            return False, "ошибка: на этой клетке стоит ваша кровать"

        if self.get_unit_at(new_x, new_y) is not None:
            return False, "ошибка: клетка уже занята другим бойцом"

        landscape = self.grid_landscape[new_y][new_x]
        if not landscape.can_pass(unit):
            return False, f"ошибка: {unit.name} не может пройти через {landscape.name}"

        unit.x = new_x
        unit.y = new_y
        text = landscape.apply_effect(unit)

        # если на клетке есть нейтральный объект, используем перегрузку оператора +
        neutral = self.get_neutral_at(new_x, new_y)
        if neutral is not None and unit.health > 0:
            bonus_text = unit + neutral
            text = text + " | " + bonus_text
            self.remove_neutral_object(neutral)

        if unit.health <= 0:
            self.remove_unit(unit)
            text = text + f" | боец {unit.name} погиб"

        logger.log(text)
        return True, text

    def copy_field(self):
        # глубокое копирование поля вместе со всеми объектами
        logger.log("выполнено копирование игрового поля")
        return copy.deepcopy(self)

    def render(self):
        result = f"\n--- карта bedwars ({self.width}x{self.height}) | объектов на поле: {self.total_objects_count()}/{self.max_objects} ---\n"
        header = "   "
        for x in range(self.width):
            header += f"{x:2d} "
        result += header + "\n"

        for y in range(self.height):
            row_str = f"{y:2d}| "
            for x in range(self.width):
                unit = self.get_unit_at(x, y)
                neutral = self.get_neutral_at(x, y)

                if self.base is not None and self.base.x == x and self.base.y == y:
                    sym = self.base.symbol
                elif unit is not None:
                    sym = unit.symbol
                elif neutral is not None:
                    sym = neutral.symbol
                else:
                    sym = self.grid_landscape[y][x].symbol
                row_str += sym + "  "
            result += row_str + "\n"

        result += "\nобозначения на карте:\n"
        result += "база и бойцы: [B] кровать, [S] мечник, [P] крушитель, [A] лучник, [F] фаербол, [G] голем, [E] эндер\n"
        result += "ландшафт: [.] шерстяной мост, [O] обсидиан (+броня), [~] разлом бездны (-здоровье)\n"
        result += "объекты: [+] золотое яблоко, [D] алмазы, [$] изумруды, [X] динамит"
        return result

            if choice == "0":
                break

            if choice.isdigit():
                idx = int(choice) - 1
                if 0 <= idx < len(self.base.units):
                    self.control_unit(self.base.units[idx])
                else:
                    print("нет бойца с таким номером")
            else:
                print("введите цифру")

    def control_unit(self, unit):
        while unit in self.base.units:
            print(self.field.render())
            print("\nвыбранный боец:", unit.get_info())
            print("управление: [w] вверх | [s] вниз | [a] влево | [d] вправо | [i] подобрать объект | [0] назад")
            cmd = input("ваш ход: ").strip().lower()

            if cmd == "0":
                break
            elif cmd == "w":
                ok, text = unit.move(0, -1, self.field)
                print("результат:", text)
            elif cmd == "s":
                ok, text = unit.move(0, 1, self.field)
                print("результат:", text)
            elif cmd == "a":
                ok, text = unit.move(-1, 0, self.field)
                print("результат:", text)
            elif cmd == "d":
                ok, text = unit.move(1, 0, self.field)
                print("результат:", text)
            elif cmd == "i":
                obj = self.field.get_neutral_at(unit.x, unit.y)
                if obj is not None:
                    # перегрузка оператора +
                    print("взаимодействие:", unit + obj)
                    self.field.remove_neutral_object(obj)
                    if unit.health <= 0:
                        self.field.remove_unit(unit)
                        print("боец погиб")
                        break
                else:
                    print("на этой клетке нет нейтрального объекта")
            else:
                print("неизвестная команда")

import json


class BedWarsProfile:
    def __init__(self, nickname="Steve", team_color="красная", emeralds=100):
        self._nickname = nickname
        self._team_color = team_color
        self._emeralds = emeralds

    @property
    def nickname(self):
        return self._nickname

    @nickname.setter
    def nickname(self, value):
        if value != "":
            self._nickname = value

    @property
    def team_color(self):
        return self._team_color

    @team_color.setter
    def team_color(self, value):
        if value != "":
            self._team_color = value

    @property
    def emeralds(self):
        return self._emeralds

    @emeralds.setter
    def emeralds(self, value):
        if value >= 0:
            self._emeralds = value
        else:
            self._emeralds = 0

    def save_to_json(self, filename="saves/bedwars_profile.json"):
        # сохраняем сам объект и информацию о классе в json
        data = {
            "class_name": self.__class__.__name__,
            "attributes": ["nickname", "team_color", "emeralds"],
            "nickname": self._nickname,
            "team_color": self._team_color,
            "emeralds": self._emeralds
        }
        file = open(filename, "w", encoding="utf-8")
        json.dump(data, file, ensure_ascii=False, indent=4)
        file.close()

    def load_from_json(self, filename="saves/bedwars_profile.json"):
        try:
            file = open(filename, "r", encoding="utf-8")
            data = json.load(file)
            file.close()
            self.nickname = data["nickname"]
            self.team_color = data["team_color"]
            self.emeralds = data["emeralds"]
            return True
        except:
            return False

    def get_text(self):
        return f"игрок: {self.nickname} | команда: {self.team_color} | изумруды: {self.emeralds}"
